# Example Baota/Nginx site configuration for LNDX.
# Uploaded identity-card and insurance assets are private and must be proxied
# to the backend so Express authMiddleware can verify the token.

server {
    listen 80;
    server_name ln.tuojiayi.com;

    root /www/wwwroot/lndx/frontend/dist;
    index index.html index.htm;

    location /uploads/ {
        proxy_pass http://127.0.0.1:3000/uploads/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header Authorization $http_authorization;

        proxy_buffering off;
        add_header Cache-Control "private, no-store";
    }

    location /api/ {
        proxy_pass http://127.0.0.1:3000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    location / {
        try_files $uri $uri/ /index.html;
    }
}
